# 기획서: 영화 티어리스트 제작 도구 (Movie Tier List Maker)

## 1. 프로젝트 개요
사용자가 자신이 본 영화들을 등급(Tier)별로 분류하고 시각적으로 정리할 수 있는 단일 페이지 웹 애플리케이션을 제작한다.

## 2. 주요 기능
- **영화 추가**: 영화 제목을 입력하여 리스트에 추가하는 기능.
- **드래그 앤 드롭 (Drag & Drop)**: 추가된 영화 항목을 S, A, B, C 등급 칸으로 자유롭게 이동시키는 기능.
- **항목 삭제**: 잘못 추가하거나 제외하고 싶은 영화를 삭제하는 기능.
- **반응형 디자인**: 모바일과 데스크톱 환경에서 모두 보기 편한 레이아웃.

## 3. 티어 구성
- **S (God Tier)**: 인생 영화, 최고의 작품
- **A (Amazing)**: 주변에 적극 추천할 만한 작품
- **B (Good)**: 재미있게 본 작품
- **C (Meh)**: 평범하거나 아쉬운 작품

## 4. 기술 스택
- **Frontend**: HTML5, CSS3, JavaScript (Vanilla JS)
- **Library**: SortableJS (드래그 앤 드롭 구현을 위해 사용 권장)

## 5. 디자인 컨셉
- 어두운 배경의 다크 모드(Dark Mode) 테마.
- 각 티어별로 고유한 색상 부여 (예: S-빨강, A-주황, B-노랑 등).
- 영화 카드는 깔끔한 박스 형태로 구성.

    <script>
        // 데이터 저장 및 로드
        let movieData = JSON.parse(localStorage.getItem('myMovies')) || [];

        const input = document.getElementById('movie-input');
        const addBtn = document.getElementById('add-btn');

        // 영화 추가 함수
        addBtn.onclick = () => {
            if (!input.value) return;
            const newMovie = {
                id: Date.now(),
                title: input.value,
                stars: 5,
                tier: 'pool'
            };
            movieData.push(newMovie);
            input.value = '';
            render();
        };

        // 드래그 앤 드롭 설정
        function allowDrop(ev) { ev.preventDefault(); }
        function drag(ev) { ev.dataTransfer.setData("text", ev.target.id); }
        function drop(ev) {
            ev.preventDefault();
            const id = ev.dataTransfer.getData("text");
            const targetTier = ev.target.closest('.tier-content, .movie-pool').id;
            
            const movie = movieData.find(m => m.id == id);
            if (movie) {
                movie.tier = targetTier.replace('tier-', '');
                render();
            }
        }

        // 삭제 함수
        function deleteMovie(id) {
            movieData = movieData.filter(m => m.id !== id);
            render();
        }

        // 별점 변경 함수
        function changeStars(id) {
            const movie = movieData.find(m => m.id == id);
            movie.stars = (movie.stars % 5) + 1;
            render();
        }

        // 화면 그리기
        function render() {
            localStorage.setItem('myMovies', JSON.stringify(movieData));
            
            // 모든 구역 비우기
            document.querySelectorAll('.tier-content, .movie-pool').forEach(el => el.innerHTML = '');

            movieData.forEach(movie => {
                const card = document.createElement('div');
                card.className = 'movie-card';
                card.draggable = true;
                card.id = movie.id;
                card.ondragstart = drag;
                
                card.innerHTML = `
                    <button class="delete-btn" onclick="deleteMovie(${movie.id})">X</button>
                    <div>${movie.title}</div>
                    <div class="stars" onclick="changeStars(${movie.id})">${'★'.repeat(movie.stars)}${'☆'.repeat(5-movie.stars)}</div>
                `;

                const targetId = movie.tier === 'pool' ? 'movie-pool' : `tier-${movie.tier}`;
                document.getElementById(targetId).appendChild(card);
            });
        }

        render();
    </script>

    <script>
        let movieData = JSON.parse(localStorage.getItem('myMovies')) || [];

        // 1. 드래그 앤 드롭 설정 (SortableJS 사용)
        const containers = document.querySelectorAll('.tier-content');
        containers.forEach(container => {
            new Sortable(container, {
                group: 'shared', // 여러 구역 간 이동 가능하게 설정
                animation: 150,
                onEnd: saveAll // 이동이 끝나면 저장
            });
        });

        // 2. 영화 추가
        document.getElementById('add-btn').onclick = () => {
            const input = document.getElementById('movie-input');
            if (!input.value) return;
            const newMovie = { id: Date.now().toString(), title: input.value, stars: 5 };
            movieData.push(newMovie);
            addMovieToUI('pool', newMovie);
            input.value = '';
            saveAll();
        };

        // 3. UI에 영화 카드 그리기
        function addMovieToUI(tierId, movie) {
            const card = document.createElement('div');
            card.className = 'movie-card';
            card.dataset.id = movie.id;
            card.innerHTML = `
                <button class="delete-btn" onclick="removeMovie(this)">X</button>
                <div style="font-weight:bold">${movie.title}</div>
                <div class="stars" onclick="toggleStars(this)">${'★'.repeat(movie.stars)}${'☆'.repeat(5-movie.stars)}</div>
            `;
            document.getElementById(tierId).appendChild(card);
        }

        // 4. 별점 변경
        function toggleStars(el) {
            let starsText = el.innerText;
            let count = (starsText.match(/★/g) || []).length;
            count = (count % 5) + 1;
            el.innerText = '★'.repeat(count) + '☆'.repeat(5 - count);
            saveAll();
        }

        // 5. 삭제 및 초기화
        function removeMovie(btn) {
            btn.parentElement.remove();
            saveAll();
        }

        document.getElementById('reset-btn').onclick = () => {
            if(confirm("정말 모든 데이터를 지울까요?")) {
                localStorage.removeItem('myMovies');
                location.reload();
            }
        };

        // 6. 현재 상태 저장
        function saveAll() {
            const currentData = [];
            document.querySelectorAll('.tier-content').forEach(container => {
                const tierId = container.id;
                container.querySelectorAll('.movie-card').forEach(card => {
                    const starsText = card.querySelector('.stars').innerText;
                    currentData.push({
                        id: card.dataset.id,
                        title: card.querySelector('div').innerText,
                        stars: (starsText.match(/★/g) || []).length,
                        tier: tierId
                    });
                });
            });
            localStorage.setItem('myMovies', JSON.stringify(currentData));
        }

        // 초기 로드
        movieData.forEach(m => addMovieToUI(m.tier || 'pool', m));
    </script>

<script>
    /* 2. JavaScript: 기능 부분 */

    // 페이지 로드 시 데이터 불러오기
    window.onload = () => {
        loadData();
        initDragAndDrop();
    };

    // 드래그 앤 드롭 설정 (SortableJS)
    function initDragAndDrop() {
        const containers = document.querySelectorAll('.tier-content');
        containers.forEach(container => {
            new Sortable(container, {
                group: 'shared', // 여러 목록 간 이동 가능하게 설정
                animation: 150,
                ghostClass: 'blue-background-class',
                onEnd: saveData // 이동이 끝나면 자동 저장
            });
        });
    }

    // 영화 추가 함수
    function addMovie(title = null, stars = 5) {
        const input = document.getElementById('movie-input');
        const movieTitle = title || input.value.trim();

        if (!movieTitle) return;

        const card = document.createElement('div');
        card.className = 'movie-card';
        card.innerHTML = `
            <span class="movie-name">${movieTitle}</span>
            <span class="star" onclick="changeStar(this)">★${stars}</span>
            <span class="delete-btn" onclick="deleteMovie(this)">X</span>
        `;

        document.getElementById('movie-pool').appendChild(card);
        if (!title) input.value = ''; // 직접 입력했을 때만 입력창 비우기
        saveData();
    }

    // 별점 변경 (1~5 순환)
    function changeStar(el) {
        let current = parseInt(el.innerText.replace('★', ''));
        let next = current >= 5 ? 1 : current + 1;
        el.innerText = `★${next}`;
        saveData();
    }

    // 개별 삭제
    function deleteMovie(el) {
        if(confirm('이 영화를 삭제할까요?')) {
            el.parentElement.remove();
            saveData();
        }
    }

    // 전체 초기화
    function resetAll() {
        if(confirm('모든 데이터를 지우고 초기화할까요?')) {
            localStorage.removeItem('movieTierData');
            location.reload();
        }
    }

    // 로컬 저장소에 데이터 저장
    function saveData() {
        const data = {};
        const containers = document.querySelectorAll('.tier-content');
        containers.forEach(container => {
            const id = container.id;
            const movies = [];
            container.querySelectorAll('.movie-card').forEach(card => {
                movies.push({
                    title: card.querySelector('.movie-name').innerText,
                    stars: card.querySelector('.star').innerText.replace('★', '')
                });
            });
            data[id] = movies;
        });
        localStorage.setItem('movieTierData', JSON.stringify(data));
    }

    // 로컬 저장소에서 데이터 불러오기
    function loadData() {
        const savedData = localStorage.getItem('movieTierData');
        if (!savedData) return;

        const data = JSON.parse(savedData);
        for (const [containerId, movies] of Object.entries(data)) {
            const container = document.getElementById(containerId);
            movies.forEach(movie => {
                const card = document.createElement('div');
                card.className = 'movie-card';
                card.innerHTML = `
                    <span class="movie-name">${movie.title}</span>
                    <span class="star" onclick="changeStar(this)">★${movie.stars}</span>
                    <span class="delete-btn" onclick="deleteMovie(this)">X</span>
                `;
                container.appendChild(card);
            });
        }
    }

<script>
    window.onload = () => {
        loadData();
        initDragAndDrop();
    };

    function initDragAndDrop() {
        const containers = document.querySelectorAll('.tier-content');
        containers.forEach(container => {
            new Sortable(container, {
                group: 'shared',
                animation: 150,
                onEnd: saveData
            });
        });
    }

    // 영화 추가
    function addMovie(title = null, rating = 0) {
        const input = document.getElementById('movie-input');
        const movieTitle = title || input.value.trim();

        if (!movieTitle) return;

        const card = document.createElement('div');
        card.className = 'movie-card';
        
        // 별 5개 생성
        let starsHtml = '';
        for(let i = 1; i <= 5; i++) {
            const activeClass = i <= rating ? 'active' : '';
            starsHtml += `<span class="star ${activeClass}" onclick="setRating(this, ${i})">★</span>`;
        }

        card.innerHTML = `
            <span class="delete-btn" onclick="deleteMovie(this)">X</span>
            <span class="movie-name">${movieTitle}</span>
            <div class="star-rating">${starsHtml}</div>
        `;

        document.getElementById('movie-pool').appendChild(card);
        if (!title) input.value = '';
        saveData();
    }

    // 별점 설정 함수
    function setRating(el, score) {
        const ratingContainer = el.parentElement;
        const stars = ratingContainer.querySelectorAll('.star');
        
        stars.forEach((star, index) => {
            if (index < score) {
                star.classList.add('active');
            } else {
                star.classList.remove('active');
            }
        });
        saveData();
    }

    function deleteMovie(el) {
        if(confirm('삭제하시겠습니까?')) {
            el.parentElement.remove();
            saveData();
        }
    }

    function resetAll() {
        if(confirm('모든 데이터를 초기화할까요?')) {
            localStorage.removeItem('movieTierDataV2');
            location.reload();
        }
    }

    function saveData() {
        const data = {};
        document.querySelectorAll('.tier-content').forEach(container => {
            const movies = [];
            container.querySelectorAll('.movie-card').forEach(card => {
                const activeStars = card.querySelectorAll('.star.active').length;
                movies.push({
                    title: card.querySelector('.movie-name').innerText,
                    rating: activeStars
                });
            });
            data[container.id] = movies;
        });
        localStorage.setItem('movieTierDataV2', JSON.stringify(data));
    }

    function loadData() {
        const savedData = localStorage.getItem('movieTierDataV2');
        if (!savedData) return;

        const data = JSON.parse(savedData);
        for (const [containerId, movies] of Object.entries(data)) {
            const container = document.getElementById(containerId);
            movies.forEach(movie => {
                addMovieToContainer(container, movie.title, movie.rating);
            });
        }
    }

    // 로드용 보조 함수
    function addMovieToContainer(container, title, rating) {
        const card = document.createElement('div');
        card.className = 'movie-card';
        let starsHtml = '';
        for(let i = 1; i <= 5; i++) {
            const activeClass = i <= rating ? 'active' : '';
            starsHtml += `<span class="star ${activeClass}" onclick="setRating(this, ${i})">★</span>`;
        }
        card.innerHTML = `
            <span class="delete-btn" onclick="deleteMovie(this)">X</span>
            <span class="movie-name">${title}</span>
            <div class="star-rating">${starsHtml}</div>
        `;
        container.appendChild(card);
    }

	•			기존: ★5 처럼 텍스트 하나로 표시.

	•	
	•	## 1. 프로젝트 개요
	•	사용자가 자신이 본 영화들을 등급(Tier)별로 분류하고 시각적으로 정리할 수 있는 단일 페이지 웹 애플리케이션을 제작한다.
	•	
	•	## 2. 주요 기능
	•	- **영화 추가**: 영화 제목을 입력하여 리스트에 추가하는 기능.
	•	- **드래그 앤 드롭 (Drag & Drop)**: 추가된 영화 항목을 S, A, B, C 등급 칸으로 자유롭게 이동시키는 기능.
	•	- **항목 삭제**: 잘못 추가하거나 제외하고 싶은 영화를 삭제하는 기능.
	•	- **반응형 디자인**: 모바일과 데스크톱 환경에서 모두 보기 편한 레이아웃.
	•	
	•	## 3. 티어 구성
	•	- **S (God Tier)**: 인생 영화, 최고의 작품
	•	- **A (Amazing)**: 주변에 적극 추천할 만한 작품
	•	- **B (Good)**: 재미있게 본 작품
	•	- **C (Meh)**: 평범하거나 아쉬운 작품
	•	
	•	## 4. 기술 스택
	•	- **Frontend**: HTML5, CSS3, JavaScript (Vanilla JS)
	•	- **Library**: SortableJS (드래그 앤 드롭 구현을 위해 사용 권장)
	•	
	•	## 5. 디자인 컨셉
	•	- 어두운 배경의 다크 모드(Dark Mode) 테마.
	•	- 각 티어별로 고유한 색상 부여 (예: S-빨강, A-주황, B-노랑 등).
	•		•	- 영화 카드는 깔끔한 박스 형태로 구성.

텍스트 복사도 좋지만, **"과정"**을 보여주기에는 PDF 저장이 가장 깔끔하고 신뢰도가 높습니다. 지금 바로 이렇게 해보세요:
	•			키보드에서 Ctrl + P (맥은 Command + P)를 누릅니다.
	•			인쇄 창이 뜨면 오른쪽 설정에서 [대상] 혹은 **[프린터]**를 **[PDF로 저장]**으로 바꿉니다.
	•			[저장] 버튼을 누르면 지금 이 화면(우리가 나눈 대화 전체)이 그대로 파일로 저장됩니다.
	•			이 PDF 파일을 제출하시면 교수님께서 "아, 이 학생이 AI와 이렇게 대화하며 코드를 수정해 나갔구나!" 하고 한눈에 확인하실 수 있습니다.
